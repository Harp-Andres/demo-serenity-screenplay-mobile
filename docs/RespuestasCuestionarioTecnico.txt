UPDATE 
    CAREERS
SET 
    DESCRIPTION = 'Administración de Empresas'
WHERE 
    CAREER_ID = '10026';
	
--d. Muestre los nombres de los estudiantes que estén cursando dos o más materias.

SELECT 
    S.name
FROM 
    STUDENTS S
JOIN 
    CAREER_STUDENT CS ON S.STUDENT_ID = CS.STUDENT_ID
GROUP BY 
    S.STUDENT_ID, S.name
HAVING 
    COUNT(CS.CAREER_ID) >= 2;
	
--e. Calificación promedio de todos los estudiantes que estén cursando carreras

SELECT 
    AVG(CALIFICATION) AS promedio_calificacion
FROM 
    CAREER_STUDENT;

--f. Muestra el nombre, carrera id, correos de los estudiantes que tienen una calificación mayor 70.

SELECT 
    S.name, 
    CS.CAREER_ID, 
    S.EMAIL
FROM 
    STUDENTS S
JOIN 
    CAREER_STUDENT CS ON S.STUDENT_ID = CS.STUDENT_ID
WHERE 
    CS.CALIFICATION > 70; 
	
--g. Cuántos estudiantes que estén cursando 2 o más carreras

SELECT 
    COUNT(*) AS total_estudiantes_multi_carrera
FROM (
    SELECT student_id
    FROM CAREER_STUDENT
    GROUP BY student_id
    HAVING COUNT(DISTINCT career_id) >= 2
) AS SubQuery;

/*
a) En la vista de Login se implementa un método para validar que todos los labels
definidos en el diseño se encuentren presentes. A continuación verá el código
implementado.
Instrucción: De acuerdo a su experiencia indique si está de acuerdo con la forma en la que
fue implementado el método. En caso de no estar de acuerdo. Indíquenos qué
modificaciones haría o reescriba el método para ver reflejados sus cambios y justifique su
respuesta.


public class LoginPage {
    private final AppiumDriver driver;
    private final WebDriverWait wait;

    @AndroidFindBy(id = "id_user")
    @iOSXCUITFindBy(accessibility = "id_user")
    private WebElement labelUser;

    @AndroidFindBy(xpath = "//*")
    @iOSXCUITFindBy(accessibility = "id_label_hola")
    private WebElement labelHola;
	
	@AndroidFindBy(xpath = "//*")
    @iOSXCUITFindBy(accessibility = "id_label_pass")
    private WebElement labelPass;
	
	@AndroidFindBy(xpath = "//*")
    @iOSXCUITFindBy(accessibility = "id_label_forgot")
    private WebElement labelForgot;

    public LoginPage(AppiumDriver driver) {
        this.driver = driver;
        this.wait = new WebDriverWait(driver, Duration.ofSeconds(15));
        PageFactory.initElements(new AppiumFieldDecorator(driver), this);
    }
	
	public boolean verifyLabelsArePresent(AppiumDriver driver) {
	  
		WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(15));
		
		try {
		   
			wait.until(ExpectedConditions.visibilityOf(labelUser));
            wait.until(ExpectedConditions.visibilityOf(labelHola));
			wait.until(ExpectedConditions.visibilityOf(labelPass));
			//Esta verificacion tal vez toque hacer solo en e3scenarios negativos donde el pass se ha incorrecto
            wait.until(ExpectedConditions.visibilityOf(labelForgot));

			boolean isUserCorrect = labelUser.getText().contains("Iniciar Sesión");
			boolean isHolaCorrect = labelHola.getText().contains("Hola");
			boolean isPassCorrect = labelPass.getText().equals("Ingresa tu contraseña de 6 dígitos");
			//Esta verificacion tal vez toque hacer solo en e3scenarios negativos donde el pass se ha incorrecto
			boolean isForgotCorrect = labelForgot.getText().equals("Olvidé mi contraseña");

			return isUserCorrect && isHolaCorrect && isPassCorrect && isForgotCorrect;

		} catch (Exception e) {
			System.err.println("Fallo al verificar etiquetas: " + e.getMessage());
			return false;
		}
	}
}

eliminando el ineficiente y estático Thread.sleep(). 
Además, se usan localizadores con pagefactory con anotaciones multiplataforma aplicando POM con pagefactory.
